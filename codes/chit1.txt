-- Chit 1: MySQL-compatible

DROP TABLE IF EXISTS employee_master;
DROP TABLE IF EXISTS branch_master;

CREATE TABLE branch_master (
  branch_id INT PRIMARY KEY,
  branch_name VARCHAR(100) NOT NULL
);

CREATE TABLE employee_master (
  emp_id INT PRIMARY KEY,
  first_name VARCHAR(50) NOT NULL,
  middle VARCHAR(50),
  last_name VARCHAR(50),
  department VARCHAR(50) NOT NULL,
  manager_id INT,
  CONSTRAINT fk_manager FOREIGN KEY (manager_id) REFERENCES employee_master(emp_id)
    ON DELETE SET NULL
);

-- 3. Create index on emp_name: MySQL doesn't support functional index directly, so add a generated column and index it
ALTER TABLE employee_master
  ADD COLUMN emp_name VARCHAR(200) GENERATED ALWAYS AS (CONCAT(first_name, ' ', IFNULL(middle,''), ' ', IFNULL(last_name,''))) VIRTUAL;

CREATE INDEX idx_emp_name ON employee_master (emp_name);

-- 4. Create a view containing employee details (including manager name)
DROP VIEW IF EXISTS v_employee_details;
CREATE VIEW v_employee_details AS
SELECT e.emp_id,
       e.first_name,
       e.middle,
       e.last_name,
       e.department,
       e.manager_id,
       CONCAT(m.first_name, ' ', IFNULL(m.last_name,'')) AS manager_name
FROM employee_master e
LEFT JOIN employee_master m ON e.manager_id = m.emp_id;

-- 5. Create a role and grant read-only access to the view (MySQL 8+)
-- Note: user/role creation requires appropriate privileges
CREATE ROLE IF NOT EXISTS readonly_role;
GRANT SELECT ON test.v_employee_details TO readonly_role; -- change schema 'test' to your DB name

-- 6. View listing each manager and their direct report count
DROP VIEW IF EXISTS v_manager_reports;
CREATE VIEW v_manager_reports AS
SELECT m.emp_id AS manager_id,
       CONCAT(m.first_name, ' ', IFNULL(m.last_name,'')) AS manager_name,
       COUNT(e.emp_id) AS direct_reports
FROM employee_master e
JOIN employee_master m ON e.manager_id = m.emp_id
GROUP BY m.emp_id, m.first_name, m.last_name;

-- 7. Prevent exact duplicate people within the same department
ALTER TABLE employee_master
  ADD CONSTRAINT uq_person_dept UNIQUE (first_name, middle, last_name, department);

-- 8. Add CHECK constraint to restrict departments to a known list (MySQL 8 enforces CHECK)
ALTER TABLE employee_master
  ADD CONSTRAINT chk_department CHECK (department IN ('IT','HR','Finance','Sales','Admin'));
