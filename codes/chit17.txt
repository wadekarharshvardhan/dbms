-- Chit 17: SALES department 10% hike + logging + constraints

DROP TABLE IF EXISTS EMP;
DROP TABLE IF EXISTS salary_log;

CREATE TABLE EMP (
  emp_id INT PRIMARY KEY,
  emp_name VARCHAR(100),
  dept VARCHAR(50),
  salary DECIMAL(12,2)
);

INSERT INTO EMP VALUES (1,'A', 'SALES', 80000), (2,'B','SALES', 120000), (3,'C','DEV', 70000);

CREATE TABLE salary_log (
  log_id INT AUTO_INCREMENT PRIMARY KEY,
  emp_id INT,
  old_salary DECIMAL(12,2),
  new_salary DECIMAL(12,2),
  changed_on TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  reason VARCHAR(200)
);

-- Start transaction, update sales by 10% but only for salary <= 100000
START TRANSACTION;
UPDATE EMP SET salary = salary * 1.10 WHERE dept = 'SALES' AND salary <= 100000;
SELECT * FROM EMP WHERE dept = 'SALES';
ROLLBACK; -- undo

-- Repeat and commit
START TRANSACTION;
UPDATE EMP SET salary = salary * 1.10 WHERE dept = 'SALES' AND salary <= 100000;
COMMIT;

-- Use savepoint example
START TRANSACTION;
UPDATE EMP SET salary = salary * 1.10 WHERE emp_id = 1;
UPDATE EMP SET salary = salary * 1.10 WHERE emp_id = 2;
SAVEPOINT after_two;
UPDATE EMP SET salary = salary * 1.10 WHERE emp_id = 3;
ROLLBACK TO SAVEPOINT after_two;
COMMIT;

-- 6. Trigger to log salary changes (before update)
DELIMITER $$
CREATE TRIGGER trg_salary_log BEFORE UPDATE ON EMP
FOR EACH ROW
BEGIN
  IF NEW.salary <> OLD.salary THEN
    INSERT INTO salary_log(emp_id, old_salary, new_salary, reason)
    VALUES (OLD.emp_id, OLD.salary, NEW.salary, 'Hike applied');
  END IF;
END$$
DELIMITER ;

-- 7. Prevent salary hike if employee’s salary already exceeds 100000
-- Enforced in UPDATE WHERE clause. Alternatively create trigger to block:
DELIMITER $$
CREATE TRIGGER trg_prevent_high_salary BEFORE UPDATE ON EMP
FOR EACH ROW
BEGIN
  IF NEW.salary > OLD.salary AND OLD.salary > 100000 THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Salary hike not allowed for salary > 100000';
  END IF;
END$$
DELIMITER ;

-- 8. Display count of employees affected by last update
-- After running an update, read ROW_COUNT() in your client. Example:
SELECT COUNT(*) FROM EMP WHERE dept='SALES' AND salary <= 110000; -- approximation

-- 9. Stored procedure proc_apply_hike(dept_name, hike_percent)
DELIMITER $$
CREATE PROCEDURE proc_apply_hike(IN p_dept_name VARCHAR(50), IN p_hike_percent DECIMAL(5,2))
BEGIN
  START TRANSACTION;
  UPDATE EMP
  SET salary = salary * (1 + p_hike_percent/100)
  WHERE dept = p_dept_name AND salary <= 100000;
  SELECT ROW_COUNT() AS rows_updated;
  COMMIT;
END$$
DELIMITER ;
